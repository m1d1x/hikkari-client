cmake_minimum_required(VERSION 3.14)
project(runner LANGUAGES CXX)

# Define the application target. To change its name, change BINARY_NAME in the
# top-level CMakeLists.txt, not the value here, or `flutter run` will no longer
# work.
#
# Any new source files that you add to the application should be added here.
add_executable(${BINARY_NAME} WIN32
  "flutter_window.cpp"
  "main.cpp"
  "utils.cpp"
  "win32_window.cpp"
  "${FLUTTER_MANAGED_DIR}/generated_plugin_registrant.cc"
  "Runner.rc"
)
# Note: runner.exe.manifest is intentionally NOT in the source list.
# It's embedded via Runner.rc (resource type 24, ID 1) instead of being
# processed by MSBuild's ManifestTool — see the comment in Runner.rc.

# Apply the standard set of build settings. This can be removed for applications
# that need different build settings.
apply_standard_settings(${BINARY_NAME})

# Tell the linker NOT to generate its own manifest. Our manifest is baked
# into the binary as a Win32 resource via Runner.rc (1 24 "runner.exe.manifest"),
# so there's nothing for the linker / mt.exe to do separately. Without
# this, the default /MANIFEST flag still triggers the ManifestTool step
# and we hit the same LNK1327 mt.exe race against Windows Defender.
target_link_options(${BINARY_NAME} PRIVATE "/MANIFEST:NO")

# Add preprocessor definitions for the build version.
target_compile_definitions(${BINARY_NAME} PRIVATE "FLUTTER_VERSION=\"${FLUTTER_VERSION}\"")
target_compile_definitions(${BINARY_NAME} PRIVATE "FLUTTER_VERSION_MAJOR=${FLUTTER_VERSION_MAJOR}")
target_compile_definitions(${BINARY_NAME} PRIVATE "FLUTTER_VERSION_MINOR=${FLUTTER_VERSION_MINOR}")
target_compile_definitions(${BINARY_NAME} PRIVATE "FLUTTER_VERSION_PATCH=${FLUTTER_VERSION_PATCH}")
target_compile_definitions(${BINARY_NAME} PRIVATE "FLUTTER_VERSION_BUILD=${FLUTTER_VERSION_BUILD}")

# Disable Windows macros that collide with C++ standard library functions.
target_compile_definitions(${BINARY_NAME} PRIVATE "NOMINMAX")

# Add dependency libraries and include directories. Add any application-specific
# dependencies here.
target_link_libraries(${BINARY_NAME} PRIVATE flutter flutter_wrapper_app)
target_link_libraries(${BINARY_NAME} PRIVATE "dwmapi.lib")
target_include_directories(${BINARY_NAME} PRIVATE "${CMAKE_SOURCE_DIR}")

# Run the Flutter tool portions of the build. This must not be removed.
add_dependencies(${BINARY_NAME} flutter_assemble)

# Drop every native runtime artifact next to the runner exe after every
# build. These are *not* tracked by CMake's install rules, so a fresh
# `flutter clean` would otherwise wipe them and leave the app launching
# into an empty profile / no sing-box state. Sources live in the repo:
#
#   nekkobox.dll   — built by `go build -buildmode=c-shared` at the repo root
#   sing-box.exe   — committed binary, the actual VPN engine
#   wintun.dll     — committed Wintun driver, sing-box loads this for TUN
#   config.json    — sing-box config (regenerated by the Go side on switch)
#   profiles.json  — list of user-configured profiles
#
# config.json and profiles.json are copied only if they don't already exist
# in the destination — otherwise we'd clobber the user's live state on
# every rebuild.
set(NEKKOBOX_REPO_ROOT "${CMAKE_SOURCE_DIR}/../..")
set(NEKKOBOX_BIN_SRC "${NEKKOBOX_REPO_ROOT}/NekkoVPN/build/bin")

add_custom_command(TARGET ${BINARY_NAME} POST_BUILD
  # Always-overwrite: native binaries the user shouldn't be editing.
  COMMAND ${CMAKE_COMMAND} -E copy_if_different
          "${NEKKOBOX_REPO_ROOT}/nekkobox.dll"
          "$<TARGET_FILE_DIR:${BINARY_NAME}>/nekkobox.dll"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different
          "${NEKKOBOX_BIN_SRC}/sing-box.exe"
          "$<TARGET_FILE_DIR:${BINARY_NAME}>/sing-box.exe"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different
          "${NEKKOBOX_BIN_SRC}/wintun.dll"
          "$<TARGET_FILE_DIR:${BINARY_NAME}>/wintun.dll"
  # Seed-only: never overwrite user state. We pipe through a CMake script
  # that checks existence before copying so a rebuild on a long-lived dev
  # box doesn't reset profiles / whitelist / DNS rules.
  COMMAND ${CMAKE_COMMAND}
          -DSRC=${NEKKOBOX_BIN_SRC}/config.json
          -DDST=$<TARGET_FILE_DIR:${BINARY_NAME}>/config.json
          -P ${CMAKE_CURRENT_SOURCE_DIR}/copy_if_missing.cmake
  COMMAND ${CMAKE_COMMAND}
          -DSRC=${NEKKOBOX_BIN_SRC}/profiles.json
          -DDST=$<TARGET_FILE_DIR:${BINARY_NAME}>/profiles.json
          -P ${CMAKE_CURRENT_SOURCE_DIR}/copy_if_missing.cmake
  COMMENT "Seed nekkobox runtime files → $<TARGET_FILE_DIR:${BINARY_NAME}>"
  VERBATIM
)
